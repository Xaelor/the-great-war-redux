			always = no
		}
		available = {
			hidden_trigger = {
				always = no
			}
		}
		cancel_trigger = {
			hidden_trigger = {
				OR = {

						is_prime_minister = no

					NOT = {
						has_decision = RUS_law_on_western_zemstvo_sc
					}

			}
		}

		days_mission_timeout = 248
		selectable_mission = no
		timeout_effect = {
			RUS_end_the_vote = yes
			hidden_effect = {
				add_political_power = 50
			}
			activate_decision = RUS_law_on_western_zemstvo_sc
		}
	}
