debug_decisions = {
	debug_improve_armenian_loyalty = {
		allowed = {
			tag = TUR
		}
		complete_effect = {
			add_to_variable = {
				TUR_armenian_loyalty_var = 0.05
			}
			add_to_variable = {
				TUR_armenian_loyalty_display_var = 5
				tooltip = TUR_modify_armenian_loyalty_tt
			}
			clamp_variable = {
				var = TUR_armenian_loyalty_var
				min = -1
				max = 1
			}
			TUR_recalculate_millet_system_modifiers = yes
		}
	}
	debug_remove_armenians = {
		allowed = {
			tag = TUR
		}
		complete_effect = {
			352 = {
				set_temp_variable = {
					TUR_modify_armenian_pop_temp_var = {
						value = THIS.TUR_armenian_state_pop_var
						multiply = -0.23
					}
				}
				set_temp_variable = {
					TUR_modify_armenian_state_pop_temp_var = {
						value = TUR_modify_armenian_pop_temp_var
						multiply = 1000000
					}
				}
				set_temp_variable = {
					TUR_modify_state_pop_temp_var = {
						value = 0
						add = TUR_modify_armenian_state_pop_temp_var
					}
				}
				add_manpower = TUR_modify_state_pop_temp_var
				add_to_variable = {
					THIS.TUR_armenian_state_pop_var = TUR_modify_armenian_pop_temp_var
					tooltip = TUR_modify_armenian_pop_tt
				}
			}
			TUR_recalculate_millet_system_modifiers = yes
		}
	}
	debug_improve_centralization = {
		allowed = {
			tag = TUR
		}
		complete_effect = {
			every_owned_state = {
				limit = {
					check_variable = {
						THIS.TUR_muslim_state_var = 1
					}
				}
				add_to_variable = {
					THIS.TUR_state_centralization_var = 0.1
					tooltip = TUR_modify_centralization_tt
				}
			}
			hidden_effect = {
				every_owned_state = {
					limit = {
						check_variable = {
							THIS.TUR_muslim_state_var = 1
						}
					}
					clamp_variable = {
						var = THIS.TUR_state_centralization_var
						min = 0
						max = 1
					}
				}
			}
			TUR_recalculate_millet_system_modifiers = yes
		}
	}
}
