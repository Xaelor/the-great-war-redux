		icon = GFX_decision_generic_civil_support

		}

		available = {
			OR = {
				check_variable = {
					var = RUS_revolution_suppression_progress

				}

					has_agrarian_question = no

					limit = {
						has_dynamic_modifier = RUS_revolutionary_activity
					}
					remove_dynamic_modifier = {
						modifier = RUS_revolutionary_activity
					}
				}
				remove_ideas = RUS_general_strike
			}
		}

			custom_effect_tooltip = RUS_revolution_will_escalate_tt
			hidden_effect = {
				RUS_petrograd_commune = yes
				RUS_transfer_revolutionary_states = yes

					target = PET
					size = 0.3
					army_ratio = 0.4
