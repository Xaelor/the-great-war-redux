ALB_diplomatic_victories_var = {
	base = 0
	modifier = {
		set_temp_variable = { global.ALB_diplomatic_victories_var_temp = 0 }
		any_other_country = {
			is_major = yes
			capital_scope = {
				OR = {
					is_on_continent = europe
					is_on_continent = north_america
				}
			}
			has_opinion = {
				target = PREV
				value > 80
			}
			add_to_temp_variable = { global.ALB_diplomatic_victories_var_temp = 0.005 }
		}
		add = global.ALB_diplomatic_victories_var_temp
	}
}
