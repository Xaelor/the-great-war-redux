recalculate_interest_expense_and_income = {
	set_variable = {
		interest_expense = 0
	}
	set_variable = {
		interest_income = 0
	}
	for_each_scope_loop = {
		array = THIS.lenders
		set_variable = {
			PREV.interest_to_@THIS = THIS.debt_of_@PREV
		}
		multiply_variable = {
			PREV.interest_to_@THIS = THIS.interest_rate
		}
		add_to_variable = {
			PREV.interest_expense = PREV.interest_to_@THIS
		}
		for_each_scope_loop = {
			array = THIS.debtors
			add_to_variable = {
				PREV.interest_income = THIS.interest_to_@PREV
			}
			clear_variable = THIS.interest_to_@PREV
		}
	}
}
