clr_subject_flags = {
	clr_country_flag = protectorate
	clr_country_flag = colony
	clr_country_flag = monarchist_puppet
}
# scope is the country being freed
free_subject = {
	if = {
		limit = { is_subject = yes }
		overlord = {
			set_autonomy = {
				target = PREV
				autonomy_state = autonomy_free
			}
		}		
	}
	clr_subject_flags = yes
}
free_subject_leave_faction = {
	if = {
		limit = { is_subject = yes }
		overlord = {
			set_autonomy = {
				target = PREV
				autonomy_state = autonomy_free
			}
		}		
	}
	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
	}
	clr_subject_flags = yes
}

set_autonomy_protectorate = {
	free_subject_leave_faction = yes
	set_country_flag = protectorate
	get_current_government_type = yes
	PREV = {
		set_autonomy = {
			target = PREV
			autonomous_state = autonomy_tgwr_protectorate
			end_wars = no
			end_civil_wars = no
		}
	}
	restore_previous_government_type = yes
}
