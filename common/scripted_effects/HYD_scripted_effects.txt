HYD_degrade_martial_races_theory = {
	if = {
		limit = {
			has_idea = HYD_martial_races_theory
		}
		swap_ideas = {
			remove_idea = HYD_martial_races_theory
			add_idea = HYD_martial_races_theory3
		}
	}
	else_if = {

		}

		}

		}

			remove_idea = HYD_martial_races_theory5

	}
}
