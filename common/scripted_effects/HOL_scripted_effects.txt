INS_lower_illiteracy = {
	if = {
		limit = {
			has_idea = INS_mass_illiteracy4
		}
		swap_ideas = {
			remove_idea = INS_mass_illiteracy4
			add_idea = INS_mass_illiteracy3
		}
	}
	if = {
		limit = {
			has_idea = INS_mass_illiteracy3
		}
		swap_ideas = {
			remove_idea = INS_mass_illiteracy3
			add_idea = INS_mass_illiteracy2
		}
	}
	if = {
		limit = {
			has_idea = INS_mass_illiteracy2
		}
		swap_ideas = {
			remove_idea = INS_mass_illiteracy2
			add_idea = INS_mass_illiteracy1
		}
	}
	if = {
		limit = {
			has_idea = INS_mass_illiteracy1
		}
		remove_ideas = INS_mass_illiteracy1
	}
}
