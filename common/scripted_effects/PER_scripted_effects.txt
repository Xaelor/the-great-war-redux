PER_calculate_total_foreign_influnce = {
	set_variable = { var = PER_total_foreign_influence value = 0 }
	if = {
		limit = { 
			# NOT = { 
			# 	original_tag = MAK
			# 	original_tag = MRG
			# }
			ENG = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = ENG.PER_foreign_influence }
	}
	else_if = {
		limit = { 
			ENG = {
				NOT = {
					exists = yes
					has_dynamic_modifier = PER_influence_in_persia_modifier
				}
			}		
		}
		set_variable = { ENG.PER_foreign_influence = 0 }
	}
	if = {
		limit = { 
			# NOT = { original_tag = ARB }
			RUS = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = RUS.PER_foreign_influence }
	}
	else_if = {
		limit = {
			RUS = {
				NOT = {
					exists = yes
					has_dynamic_modifier = PER_influence_in_persia_modifier
				}
			}		
		}
		set_variable = { RUS.PER_foreign_influence = 0 }
	}
	if = {
		limit = {
			# NOT = { original_tag = ARB }
			GER = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = GER.PER_foreign_influence }
	}
	else_if = {
		limit = {
			GER = {
				NOT = {
					exists = yes
					has_dynamic_modifier = PER_influence_in_persia_modifier
				}
			}		
		}
		set_variable = { GER.PER_foreign_influence = 0 }
	}
	if = {
		limit = { 
			TUR = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = TUR.PER_foreign_influence }
	}
	else_if = {
		limit = {
			TUR = {
				NOT = {
					exists = yes
					has_dynamic_modifier = PER_influence_in_persia_modifier
				}
			}		
		}
		set_variable = { TUR.PER_foreign_influence = 0 }
	}
	clamp_variable = {
		var = PER_total_foreign_influence
		min = 0
		max = 1
	}
	set_variable = {
		PER_total_independence = {
			value = 1
			subtract = PER_total_foreign_influence
			min = 0
		}
	}
}
PER_calculate_foreign_influnce_modifiers = {
	set_variable = {
		PER_political_power_gain = {
			value = PER_total_foreign_influence
			multiply = -1
		}
	}
	#set_variable = { var = PER_consumer_goods_expected_value value = PER_total_foreign_influence }
	#multiply_variable = { var = PER_consumer_goods_expected_value value = 0.1 }
	set_variable = {
		PER_civilian_intel_to_others = {
			value = PER_total_foreign_influence
			multiply = 100
		}
	}
}
PER_calculate_foreign_influnce = {
	hidden_effect = {
		every_country_with_original_tag = {
			original_tag_to_check = PER
			limit = {
				has_dynamic_modifier = PER_foreign_influence_modifier
			}
			PER_calculate_total_foreign_influnce = yes
			PER_calculate_foreign_influnce_modifiers = yes
		}
	}	
}

			value = PER_foreign_influence

		}
	}
	set_variable = {
		PER_consumer_goods_factor = {
			value = PER_foreign_influence
			multiply = -0.1
		}
	}
}

		division_template = {
			name = "Berīgād-e Markazī" #Central Brigade
			division_names_group = PER_INF_01
			regiments = {
				infantry = { x = 0 y = 0 }
				infantry = { x = 0 y = 1 }
			}
			support = {
				field_gun = { x = 0 y = 0 }
				recon = { x = 0 y = 1 }
			}
		}
	}
	if = {
		limit = {
			NOT = { has_template = "Ataryād-e Qazāqkhāneh" }
		}
		division_template = {
			name = "Ataryād-e Qazāqkhāneh" #Cossack Regiment
			division_names_group = PER_COS_01
			override_model = RUS_cavalry_entity
			template_counter = 106
			regiments = {
				cavalry = { x = 0 y = 0 }
				infantry = { x = 1 y = 0 }
			}
			support = {
				field_gun = { x = 0 y = 0 }
			}
			is_locked = yes
		}
		set_division_template_cap = { 
			division_template = "Ataryād-e Qazāqkhāneh" 
			division_cap = global.PER_cossack_cap
		}
	}
	if = {
		limit = {
			NOT = { has_template = "Nīrū-ye Eshāyer" }
		}
		division_template = {
			name = "Nīrū-ye Eshāyer" # Irregular Troops
			#division_names_group = PER_INF_01
			override_model = middle_eastern_gfx_cavalry_entity
			regiments = {
				irregular_cavalry = { x = 0 y = 0 }
				irregular_infantry = { x = 1 y = 0 }
			}
			is_locked = yes
			priority = 0
		}
		set_division_template_cap = {
			division_template = "Nīrū-ye Eshāyer"
			division_cap = PER_irregulars_cap
		}
	}
	if = {
		limit = {
			has_completed_focus = PER_create_the_government_gendarmerie

				militia = { x = 0 y = 0 }
			}

			division_cap = global.PER_gendarmerie_cap
		}
