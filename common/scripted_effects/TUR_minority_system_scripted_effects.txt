TUR_south_slav_total_pop = {
	hidden_effect = {
		set_variable = { TUR_south_slav_total_pop_var = 0 }
		every_owned_state = {
			limit = {
				check_variable = { THIS.TUR_south_slav_state_pop_var > 0 }
			}
			add_to_variable = {
				TUR.TUR_south_slav_total_pop_var = TUR_south_slav_state_pop_var
			}
		}
	}
}

TUR_recalculate_millet_system_modifiers = {
	hidden_effect = {
		set_variable = {
			TUR_south_slav_loyalty_stability_var = {
				value = TUR_south_slav_pop_percentage_var
				multiply = TUR.TUR_south_slav_loyalty_var
			}
		}
		set_variable = {
			TUR_greek_loyalty_stability_var = {
				value = TUR_greek_pop_percentage_var

			}
		}
		set_variable = {
			TUR_armenian_loyalty_stability_var = {
				value = TUR_armenian_pop_percentage_var
				multiply = TUR.TUR_armenian_loyalty_var
			}
		}
		set_variable = {
			TUR_jewish_loyalty_stability_var = {

		}
		set_variable = {

				multiply = TUR.TUR_aromanian_loyalty_var
			}
		}
		set_variable = {

				value = TUR_assyrian_pop_percentage_var

			}
		}
		set_variable = {
			TUR_albanian_loyalty_stability_var = {
				value = TUR_albanian_pop_percentage_var
				multiply = TUR.TUR_albanian_loyalty_var
			}
		}
		set_variable = {
			TUR_kurdish_loyalty_stability_var = {
				value = TUR_kurdish_pop_percentage_var
				multiply = TUR.TUR_kurdish_loyalty_var
			}
		}
		set_variable = {
			TUR_arab_loyalty_stability_var = {
				value = TUR_arab_pop_percentage_var
				multiply = TUR.TUR_arab_loyalty_var
			}
		}
		set_variable = {
			TUR_millet_system_var = {
				value = 0
				add = TUR.TUR_south_slav_loyalty_stability_var
				add = TUR.TUR_greek_loyalty_stability_var
				add = TUR.TUR_armenian_loyalty_stability_var
				add = TUR.TUR_jewish_loyalty_stability_var
				add = TUR.TUR_aromanian_loyalty_stability_var
				add = TUR.TUR_assyrian_loyalty_stability_var
				add = TUR.TUR_albanian_loyalty_stability_var
				add = TUR.TUR_kurdish_loyalty_stability_var
			}
		}
		set_variable = {
			TUR_greek_loyalty_recruitable_pop_var = {
				value = TUR_greek_pop_percentage_var
				multiply = TUR.TUR_greek_loyalty_var
			}
		}
		set_variable = {
			TUR_armenian_loyalty_recruitable_pop_var = {
				value = TUR_armenian_pop_percentage_var
				multiply = TUR.TUR_armenian_loyalty_var
			}
		}
		set_variable = {
			TUR_jewish_loyalty_recruitable_pop_var = {
				value = TUR_jewish_pop_percentage_var
				multiply = TUR.TUR_jewish_loyalty_var
			}
		}
		set_variable = {
			TUR_aromanian_loyalty_recruitable_pop_var = {
				value = TUR_aromanian_pop_percentage_var
				multiply = TUR.TUR_aromanian_loyalty_var
			}
		}
		set_variable = {
			TUR_assyrian_loyalty_recruitable_pop_var = {
				value = TUR_assyrian_pop_percentage_var
				multiply = TUR.TUR_assyrian_loyalty_var
			}
		}
		set_variable = {
			TUR_albanian_loyalty_recruitable_pop_var = {
				value = TUR_albanian_pop_percentage_var
				multiply = TUR.TUR_albanian_loyalty_var
			}
		}
		set_variable = {
			TUR_kurdish_loyalty_recruitable_pop_var = {
				value = TUR_kurdish_pop_percentage_var
				multiply = TUR.TUR_kurdish_loyalty_var
			}
		}
		set_variable = {
			TUR_arab_loyalty_recruitable_pop_var = {
				value = TUR_arab_pop_percentage_var
				multiply = TUR.TUR_arab_loyalty_var
			}
		}
		if = {
			limit = {
				has_country_flag = TUR_christians_exempt_from_military_service
			}
			set_variable = {
				TUR_millet_system_recruitable_population_var = {
					value = 0
					subtract = TUR_south_slav_pop_percentage_var
					subtract = TUR_greek_pop_percentage_var
					subtract = TUR_armenian_pop_percentage_var
					subtract = TUR_jewish_pop_percentage_var
					subtract = TUR_aromanian_pop_percentage_var
					subtract = TUR_assyrian_loyalty_recruitable_pop_var
					add = TUR.TUR_albanian_loyalty_recruitable_pop_var
					add = TUR.TUR_kurdish_loyalty_recruitable_pop_var
					add = TUR.TUR_arab_loyalty_recruitable_pop_var
				}
			}
		}
		else = {
			set_variable = {
				TUR_millet_system_recruitable_population_var = {
					value = 0
					add = TUR.TUR_arab_loyalty_stability_var
					add = TUR.TUR_greek_loyalty_recruitable_pop_var
					add = TUR.TUR_armenian_loyalty_recruitable_pop_var
					add = TUR.TUR_jewish_loyalty_recruitable_pop_var
					add = TUR.TUR_aromanian_loyalty_recruitable_pop_var
					add = TUR.TUR_assyrian_loyalty_recruitable_pop_var
					add = TUR.TUR_albanian_loyalty_recruitable_pop_var
					add = TUR.TUR_kurdish_loyalty_recruitable_pop_var
					add = TUR.TUR_arab_loyalty_recruitable_pop_var
				}
			}
		}
		set_variable = {
			TUR_south_slav_loyalty_display_var = {
				value = TUR_south_slav_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_greek_loyalty_display_var = {
				value = TUR_greek_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_armenian_loyalty_display_var = {
				value = TUR_armenian_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_jewish_loyalty_display_var = {
				value = TUR_jewish_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_aromanian_loyalty_display_var = {
				value = TUR_aromanian_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_assyrian_loyalty_display_var = {
				value = TUR_assyrian_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_albanian_loyalty_display_var = {
				value = TUR_albanian_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_kurdish_loyalty_display_var = {
				value = TUR_kurdish_loyalty_var
				multiply = 100
			}
		}
		set_variable = {
			TUR_arab_loyalty_display_var = {
				value = TUR_arab_loyalty_var
				multiply = 100
			}
		}
	}
