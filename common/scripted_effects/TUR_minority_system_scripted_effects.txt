TUR_recalculate_total_vilayet_pop = {
	hidden_effect = {
		every_owned_state = {
			limit = {
				is_core_of = ROOT
				has_variable = THIS.TUR_state_centralization_var
			}
			set_variable = {
				THIS.TUR_vilayet_population_var = 0
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = ROOT
				has_variable = THIS.TUR_vilayet_var
			}
			set_variable = {
				THIS.TUR_local_state_population_var = THIS.state_population_k
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = ROOT
				has_variable = THIS.TUR_vilayet_var
			}
			add_to_variable = {
				var:THIS.TUR_vilayet_var:TUR_vilayet_population_var = THIS.TUR_local_state_population_var
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = ROOT
				has_variable = THIS.TUR_state_centralization_var
			}
			divide_variable = {
				THIS.TUR_vilayet_population_var = 1000
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = ROOT
				has_variable = THIS.TUR_vilayet_var
			}
			set_variable = {
				var:THIS.TUR_vilayet_var:TUR_south_slav_vilayet_pop_var = 0
			}
			set_variable = {
				var:THIS.TUR_vilayet_var:TUR_greek_vilayet_pop_var = 0

			}

				var:THIS.TUR_vilayet_var:TUR_greek_vilayet_pop_var = THIS.TUR_greek_state_pop_var
			}
			add_to_variable = {

TUR_recalculate_total_centalization = {
	hidden_effect = {
		set_variable = {
			TUR_vilayet_capital_total_var = 0
		}
		set_variable = {
			TUR_total_centralization_var = 0
		}
		every_owned_state = {
			limit = {
				is_core_of = ROOT
				has_variable = TUR_state_centralization_var
			}
			add_to_variable = {
				TUR.TUR_vilayet_capital_total_var = 1
			}
			add_to_variable = {
				TUR.TUR_total_centralization_var = THIS.TUR_state_centralization_var
			}
		}
		divide_variable = {
			TUR_total_centralization_var = TUR_vilayet_capital_total_var
		}
	}
}
