great_war_trigger = {
	custom_trigger_tooltip = {
		tooltip = great_war_trigger_tt
		ENT = {
			has_war = yes
			has_faction_template = faction_template_entente
			any_enemy_country = {	
				has_faction_template = faction_template_central_powers
			}
		}
	}
}

is_part_of_great_war = {
	custom_trigger_tooltip = {
		tooltip = is_part_of_great_war_tt
		has_war = yes
		OR = {
			has_faction_template = faction_template_entente
			has_faction_template = faction_template_central_powers
		}
		any_enemy_country = {	
			OR = {
				has_faction_template = faction_template_entente
				has_faction_template = faction_template_central_powers
			}
		}
	}
}
