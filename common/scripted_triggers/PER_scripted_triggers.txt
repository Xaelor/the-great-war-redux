has_PER_qajar = {
	custom_trigger_tooltip = {
		tooltip = has_PER_qajar_tt	
		OR = {
			has_country_leader = {
				character = PER_ahmad_shah_qajar
			}
			has_country_leader = {
				character = PER_mohammad_ali_shah
			}
			has_country_leader = {
				character = PER_abulfath_mirza
			}
			has_country_leader = {
				character = PER_abdol_majid_mirza
			}
		}
	}
}
