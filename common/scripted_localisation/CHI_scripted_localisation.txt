defined_text = {
	name = GetJapForeignInfluenceChi
	text = {
		trigger = {
			JAP = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = CHI_influence_in_china_modifier
			}
		}
		localization_key = CHI_JAP_foreign_influence
	}
}
