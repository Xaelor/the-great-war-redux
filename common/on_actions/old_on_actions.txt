on_actions = {
	
	on_startup = {
		effect = {
			every_country = {
				limit = {
					is_ai = no
					NOT = {
						any_other_country = {
							is_ai = no
						}
					}
				}
				set_global_flag = old_sp_detection_cflg
				country_event = old_settings.1 #SP Settings - TODO MP Later // Xaelor
				set_country_flag = old_host_country_cflg
			}
			GER = {
				set_country_flag = old_realism_content_country_cflg
			}
			LIB = {
				set_country_flag = old_realism_content_country_cflg
			}
			RUS = {
				set_country_flag = old_realism_content_country_cflg
			}
		}
	}

	on_release_as_free = {
		effect = {
			ROOT = {
				if = {
					limit = {
						has_global_flag = old_settings_focus_tree_enabled_cflg
						has_focus_tree = generic_focus
					}
					load_focus_tree = generic_focus_real
					mark_focus_tree_layout_dirty = yes
					else_if = {
						limit = {
							NOT = {
								has_global_flag = old_settings_focus_tree_enabled_cflg
							}
							has_focus_tree = generic_focus_real
						}
						load_focus_tree = generic_focus
						mark_focus_tree_layout_dirty = yes
					}
				}
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			ROOT = {
				if = {
					limit = {
						has_global_flag = old_settings_focus_tree_enabled_cflg
						has_focus_tree = generic_focus
					}
					load_focus_tree = generic_focus_real
					mark_focus_tree_layout_dirty = yes
					else_if = {
						limit = {
							NOT = {
								has_global_flag = old_settings_focus_tree_enabled_cflg
							}
							has_focus_tree = generic_focus_real
						}
						load_focus_tree = generic_focus
						mark_focus_tree_layout_dirty = yes
					}
				}
			}
		}
	}

	on_liberate = {
		effect = {
			ROOT = {
				if = {
					limit = {
						has_global_flag = old_settings_focus_tree_enabled_cflg
						has_focus_tree = generic_focus
					}
					load_focus_tree = generic_focus_real
					mark_focus_tree_layout_dirty = yes
					else_if = {
						limit = {
							NOT = {
								has_global_flag = old_settings_focus_tree_enabled_cflg
							}
							has_focus_tree = generic_focus_real
						}
						load_focus_tree = generic_focus
						mark_focus_tree_layout_dirty = yes
					}
				}
			}
		}
	}

}
