add_namespace = ww1_kongo

# Death of Pedro VI
country_event = {
	id = ww1_kongo.1
	title = ww1_kongo.1.t
	desc = ww1_kongo.1.d
	picture = GFX_report_event_african_soldiers

	is_triggered_only = yes

	trigger = {
		has_character = KOG_pedro_vi
	}

	immediate = {
		hidden_effect = {
			KOG_manuel_nkomba = {
				promote_character = absolute_monarchy
				promote_character = reactionism
			}
		}	
	}

	option = {
		name = ww1_kongo.1.a
		retire_character = KOG_pedro_vi
		effect_tooltip = {
			promote_character = KOG_manuel_nkomba
		}
	}
}
