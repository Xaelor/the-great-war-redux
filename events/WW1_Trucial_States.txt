add_namespace = ww1_trucial_states

# Death of Tahnoun bin Zayed Al Nahyan
country_event = {
	id = ww1_trucial_states.1
	title = ww1_trucial_states.1.t
	desc = ww1_trucial_states.1.d
	picture = GFX_report_event_royal_parade
	
	fire_only_once = yes
	is_triggered_only = yes
	trigger = {
		has_character = ABU_tahnoun_bin_zayed_al_nahyan 
	}
	immediate = {
		hidden_effect = {
			ABU_hamdan_bin_zayed_al_nahyan = {
				promote_character = absolute_monarchy
				promote_character = reactionism
			}
		}
	}
	
	option = {
		name = ww1_trucial_states.1.a
		retire_character = ABU_tahnoun_bin_zayed_al_nahyan
		effect_tooltip = {
			promote_character = {
				character = ABU_hamdan_bin_zayed_al_nahyan
				ideology = absolute_monarchy
			}
		}
	}
}
