add_namespace = ww1_papal

# DEATH OF PIUS X
####################
news_event = {
	id = ww1_papal.1
	title = ww1_papal.1.t
	desc = ww1_papal.1.d
	picture = GFX_news_event_PAP_pius_x_death
	trigger = {
		has_country_leader = {
			character = PAP_pius_x
		}
	}
	is_triggered_only = yes
	show_major = {
		has_catholicism = yes
	}
	immediate = {
		hidden_effect = {
			retire_character = PAP_pius_x
			promote_character = PAP_college_of_cardinals
			news_event = {
				id = ww1_papal.2
				days = 14
			}
		}
	}
	option = {
		name = ww1_papal.1.a
		trigger = {
			NOT = { has_government = communism }
		}
	}
	option = {
		name = ww1_papal.1.b
		trigger = {
			has_government = communism
		}
	}
}
