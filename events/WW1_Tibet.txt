		trigger = {
			has_global_flag = simla_convention_reject
		}
	}
	desc = {
		text = ww1_tibet.5.e
		trigger = {
			NOT = { has_global_flag = simla_convention_reject }
		}
	}
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_global_flag = simla_convention_reject }

			}
		}
	}
	########################### ACCEPT ###########################
	option = {

			has_global_flag = simla_convention_reject

		}

				add_state_claim = 434
			}
		}

		name = ww1_tibet.5.a # TIB
		trigger = {
			has_global_flag = simla_convention_reject

		}
		effect_tooltip = {
			RAJ = {
				add_state_claim = 434
			}
		}
		hidden_effect = {

			}
		}

		name = ww1_tibet.5.a # RAJ
		trigger = {
			has_global_flag = simla_convention_reject

		}

		name = ww1_tibet.5.b

		}

		transfer_state = 434
		give_guarantee = TIB
		give_market_access = TIB
	}
	option = {
		name = ww1_tibet.5.b
		trigger = {
			NOT = {
				has_global_flag = simla_convention_reject
			}
			NOT = {
				tag = RAJ
				tag = TIB
			}
		}
		give_guarantee = TIB
		give_market_access = TIB
		diplomatic_relation = {
			country = TIB
			relation = non_aggression_pact
			active = yes
		}
		every_state = {
			limit = {
				is_core_of = TIB
				is_claimed_by = ROOT
				NOT = {
					state = 1220
				}
			}
			remove_claim_by = ROOT
		}
	}
	option = {
		name = ww1_tibet.5.b
		trigger = {
			NOT = {
				has_global_flag = simla_convention_reject
			}
			tag = TIB
		}

		RAJ = {
			country_event = ww1_tibet.5
		}
	}
	########################### END ###########################
}


# Tibet Is Attacked, but China recognises as their Territory
country_event = {
	id = ww1_tibet.6
	title = ww1_tibet.6.t
	desc = ww1_tibet.6.d
	picture = GFX_report_event_china_map
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = ww1_tibet.6.a
		declare_war_on = {
			target = FROM
			type = annex_everything
		}
		every_other_country = {
			limit = {
				is_china_or_warlord = yes
				is_ally_with = ROOT
			}
			add_to_war = {
				targeted_alliance = ROOT
				enemy = FROM
			}
		}
		FROM = {
			every_enemy_country = {
				limit = {
					is_china_or_warlord = no
					NOT = {
						has_war_with = ROOT
					}
				}
				ROOT = {
					declare_war_on = {
						target = PREV
						type = annex_everything
					}
				}
				every_other_country = {
					limit = {
						is_china_or_warlord = yes
						is_ally_with = ROOT
					}
					add_to_war = {
						targeted_alliance = ROOT
						enemy = PREV
					}
				}
			}
		}
		ai_chance = {
			base = 5
			modifier = {
				add = 10
				has_war_support > 0.7
			}
			modifier = {
				add = 5
				is_major = yes
			}
			modifier = {
				add = 5
				threat > 0.5
			}
		}
	}
	option = {
		name = ww1_tibet.6.b
		add_political_power = -75
		every_possible_country = {
			limit = {
				is_china_or_warlord = yes
				TIB = {
					any_core_state = {
						is_claimed_by = PREV.PREV
						NOT = {
							state = 1220
						}
					}
				}
			}
			TIB = {
				every_core_state = {
					limit = {
						is_claimed_by = PREV.PREV
						NOT = {
							state = 1220
						}
					}
					remove_claim_by = PREV
				}
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 10
				has_war_support < 0.3
			}
			modifier = {
				add = 10
				has_stability < 0.3
			}
		}
	}
}
