# White Australia Returns to the Royal Crown
country_event = {
	id = ww1_australia.25
	title = ww1_australia.25.t
	desc = ww1_australia.25.t
	picture = GFX_report_event_generic_peaceful_annexation
	trigger = {
		original_tag = AST
		NOT = {
			has_government = communism
			has_country_leader_ideology = national_syndicalism
		}
		has_global_flag = british_civil_war
		has_civil_war = no
		has_capitulated = no
		NOT = {
			any_enemy_country = {
				original_tag = AST
			}
		}
		ENG = {
			has_civil_war = no
			NOT = {
				has_government = communism
				has_country_leader_ideology = national_syndicalism
				any_enemy_country = {
					original_tag = ENG
				}
			}
		}
	}
	fire_only_once = yes
	option = {
		name = ww1_australia.25.a
		effect_tooltip = {
			ENG = {
				set_autonomy = {
					target = ROOT
					autonomy_state = autonomy_tgwr_dominion
				}
			}
		}
		hidden_effect = {
			get_current_government_type = yes
			set_country_flag = colony
			ENG = {
				set_autonomy = {
					target = ROOT
					autonomy_state = autonomy_tgwr_dominion
					end_civil_wars = no
					end_wars = no
				}
			}
			restore_previous_government_type = yes
		}
	}
}
