add_namespace = ww1_mount_athos

# Mount Athos under our control
####################
country_event = {
	id = ww1_mount_athos.1
	title = ww1_mount_athos.1.t
	desc = ww1_mount_athos.1.d
	picture = GFX_report_event_ATH_greek_soldiers
	trigger = {
		owns_state = 345
		NOT = {
			original_tag = ATH
			has_subject = ATH 
		}
		has_war = no
		NOT = {
			AND = {
				original_tag = FRA
				345 = {
					has_state_flag = french_salonika
				}			
			}		
		}
	}
	immediate = {
		hidden_effect = {
			ATH = {
				if = {
					limit = { is_subject = yes }
					OVERLORD = {
						country_event = {
							id = ww1_mount_athos.3
							hours = 1
						}	
					}
				}
				set_country_flag = protectorate
				get_current_government_type = yes
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_tgwr_feudal_autonomy
						freedom_level = 0.5
					}	
				}
				restore_previous_government_type = yes
				country_event = ww1_mount_athos.2
			}
		}
	}
	mean_time_to_happen = { days = 0 }
	option = {
		name = ww1_mount_athos.1.a
		effect_tooltip = {
			puppet = ATH
		}
	}
}

# Death of Joachim III
####################
country_event = {
	id = ww1_mount_athos.8
	title = ww1_mount_athos.8.t
	desc = ww1_mount_athos.8.d
	picture = GFX_report_event_ATH_greek_soldiers

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_character = ATH_joachim_iii
	}

	immediate = {
		hidden_effect = {
			ATH_holy_and_sacred_synod = {
				promote_character = theocracy_ideology
				promote_character = fundamentalism
			}			
		}
	}
	
	option = {
		name = ww1_mount_athos.8.a
		retire_character = ATH_joachim_iii
		hidden_effect = {
			news_event = { id = ww1_mount_athos.9 hours = 1 }
			country_event = { id = ww1_mount_athos.10 days = 89 }		
		}	
	}
}
