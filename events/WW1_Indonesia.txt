	title = ww1_indonesia.1.t
	desc = ww1_indonesia.1.d

	is_triggered_only = yes
	option = {
		name = ww1_indonesia.1.a
		if = {
			limit = {
				num_of_factories > INS_var_num_factories_needed
			}
			custom_effect_tooltip = investors_happy_tt
			add_stability = 0.025
			add_political_power = 25
			custom_effect_tooltip = num_factories_required_increased_tt
		}
		if = {
			limit = {
				num_of_factories < INS_var_num_factories_needed
			}
			custom_effect_tooltip = investors_unhappy_tt
			add_stability = -0.05
			add_political_power = -25
			custom_effect_tooltip = num_factories_required_increased_tt
		}
	}
}
