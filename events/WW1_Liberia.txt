add_namespace = ww1_liberia

#Liberia loses Great War (to GER)
country_event = {
	id = ww1_liberia.1
	title = ww1_liberia.1.t
	desc = ww1_liberia.1.d
	picture = GFX_report_event_african_soldiers
	is_triggered_only = yes
	option = {
		name = ww1_liberia.1.a
		GER = {
			set_autonomy = {
			    target = LIB
			    autonomy_state = autonomy_tgwr_client_state
			    end_wars = no
			    end_civil_wars = no
			}
		}
		if = {
			limit = {
				GER = {
					OR = {
						has_government = neutrality
						has_government = fascism
					}
				}
			}
			add_popularity = {
				ideology = neutrality
				popularity = 0.75
			}
			set_politics = {
				ruling_party = neutrality
				elections_allowed = no
			}
			promote_character = LIB_momulu_massaquoi
			LIB_momulu_massaquoi = { 
				set_character_name = "LIB_otto_i"
				remove_country_leader_trait = LIB_puppet_of_barclay
			}		
			set_cosmetic_tag = LIB_german_protectorate
		}
	}
}
#Liberia loses Great War (to FRA)

	picture = GFX_report_event_african_soldiers
	is_triggered_only = yes
	option = {
		name = ww1_liberia.2.a
		FRA = {
			transfer_state = 862
		}
		if = {
			limit = {
				ENG = { has_country_flag = great_war_winner }
			}
			ENG = {
				set_autonomy = {
				    target = LIB
				    autonomy_state = autonomy_tgwr_client_state
				    end_wars = no
				    end_civil_wars = yes
				}
			}
		}
	}
}

# Liberia puppet or not
country_event = {
	id = ww1_liberia.3
	title = ww1_liberia.3.t
	desc = ww1_liberia.3.d
	picture = GFX_report_event_african_soldiers
	trigger = {
		OR = {
			tag = FRA
			tag = GER
		}
		has_global_flag = interbellum
		has_country_flag = great_war_winner
		is_subject = no
		LIB = {
			has_war = no
			exists = yes
			has_country_flag = great_war_loser
			is_subject = no
		}
	}
	fire_only_once = yes
	option = {
		name = ww1_liberia.3.a
	}
	option = {
		name = ww1_liberia.3.b
		add_manpower = -100
		if = {
			limit = {
				tag = FRA
			}
			LIB = { country_event = ww1_liberia.2 }
			else = {
				LIB = { country_event = ww1_liberia.1 }
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
	}
}
