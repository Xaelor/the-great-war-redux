add_namespace = ww1_balkan

# forming the alliance
country_event = {
	id = ww1_balkan.1
	title = ww1_balkan.1.t
	desc = ww1_balkan.1.d
	picture = ww1_bulgaria_3
	is_triggered_only = yes
	trigger = {
		NOT = {
			has_global_flag = kis_firstbalkan_war
		}
	}
	# lets start negotiations
	option = {
		name = ww1_balkan.1.a
		if = {
			limit = { 
				tag = SER
			}
			BUL = {
				country_event = {
					id = ww1_balkan.2
					days = 1
				}
			}
		}
		else_if = {
			limit = { 
				tag = BUL
			}
			SER = {
				country_event = {
					id = ww1_balkan.2
					days = 1
				}				
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				factor = 0
				has_game_rule = {
					rule = first_balkan_war_outcome
					option = BUL_LONE_LIBERATOR
				}
			}
		}
	}
	# lone liberator
	option = {
		name = ww1_balkan.1.b
		every_country = {
			limit = {
				OR = {
					tag = SER
					tag = BUL
					tag = MNT
					tag = GRE
				}
			}
			country_event = ww1_balkan.3
			if = {
				limit = { 
					has_active_mission = BW1_war_with_turkey
				}
				add_days_mission_timeout = {
					mission = BW1_war_with_turkey
					days = 80
				}
			}
		}
		set_global_flag = balkan_league_fail
		hidden_effect = {
			BUL = {
				complete_national_focus = BUL_lone_liberator
			}
			SER = {
				complete_national_focus = SER_rely_only_on_yourself
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 50
				has_game_rule = {
					rule = first_balkan_war_outcome
					option = BUL_LONE_LIBERATOR
				}
			}
		}
	}
}

# invitation of greece and montenegro into alliance
country_event = {
	id = ww1_balkan.4
	title = ww1_balkan.4.t
	desc = ww1_balkan.4.d
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			BUL = {
				add_to_faction = ROOT
				hidden_effect = {
					country_event = {
						id = ww1_balkan.6
						days = 1
					}
				}
			}
		}
	}
	option = {
		name = ww1_balkan.4.a
		if = {
			limit = {
				original_tag = GRE
			}
			every_state = {
				limit = {
					TUR = {
						is_owned_or_subject_PREV_trigger = yes
					}
					OR = {
						state = 345
						state = 1255
						state = 826
						state = 1257
						state = 758
						state = 731
						state = 1256
						state = 803
						state = 1225
						state = 790
						state = 974
						state = 975
						state = 871
					}
				}
				add_claim_by = GRE
			}
		}
		else_if = {
			limit = {
				original_tag = MNT
			}
			every_state = {
				limit = {
					OR = {
						state = 761
						state = 848
						state = 870
					}
				}
				add_claim_by = ROOT
			}
			every_state = {
				limit = {
					OR = {
						is_claimed_by = MNT
						is_owned_by = MNT
						is_core_of = MNT
					}
				}
				remove_claim_by = SER
			}
		}
		effect_tooltip = {
			BUL = {
				add_to_faction = ROOT
			}
		}
	}
}
